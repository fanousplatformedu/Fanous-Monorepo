### Step 4 — Edit validation schema (ALL 10 fields)
**File:** `src/lib/validation/school-admin-schemas.ts`
Add `createEditSchoolMemberSchema(msg)` factory + `TEditSchoolMemberForm` type, parallel to `createAddSchoolMemberSchema`:
```ts
export const createEditSchoolMemberSchema = (msg) =>
  z.object({
    userId:   z.string().min(1, { error: () => msg("userIdRequired") }),
    firstName:z.string().min(1, { error: () => msg("firstNameRequired") }),
    lastName: z.string().min(1, { error: () => msg("lastNameRequired") }),
    email:    z.string().min(1, { error: () => msg("emailRequired") })
                 .email({ error: () => msg("emailInvalid") }),
    mobile:   z.string().min(1, { error: () => msg("mobileRequired") }),
    username: z.string()                  // optional — blank = keep existing
                 .optional(),
    password: z.string()                  // optional — blank = keep existing
                 .optional(),
    role:     z.enum(["STUDENT","PARENT","COUNSELOR","SCHOOL_ADMIN"], { error: () => msg("roleRequired") }),
    isActive: z.boolean(),
    forcePasswordChange: z.boolean(),
  });
export type TEditSchoolMemberForm = z.infer<ReturnType<typeof createEditSchoolMemberSchema>>;
```
All 10 `EditSchoolUserInput` fields are represented. Same localized zod v4 `{ error: () => msg(...) }` pattern.

- **`onSubmit` payload — ALL 10 `EditSchoolUserInput` fields**, with `username`/`password` spread in only when non-blank so the backend preserves the existing values (sending `""` would otherwise blank them):
```ts
const onSubmit = async (values: TEditSchoolMemberForm) => {
  const payload: TAPI.EditSchoolUserInput = {
    userId: member.id,
    firstName: values.firstName.trim(),
    lastName: values.lastName.trim(),
    email: values.email.trim(),
    mobile: values.mobile.trim(),
    role: values.role,
    isActive: Boolean(values.isActive),
    forcePasswordChange: Boolean(values.forcePasswordChange),
    ...(values.username.trim() ? { username: values.username.trim() } : {}),
    ...(values.password ? { password: values.password } : {}),
  };
  await editSchoolUser(payload).unwrap();   // all fields covered
  ...
};
```
